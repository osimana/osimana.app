    function setStatus(id, st) {

      const p = players.find(x => x.id === id);

      if(p) {

        p.status = st;

        if(st !== 'p') p.group = null;

      }

      render();

    }


    function assegnaGruppoManuale(id, gruppo) {

      const p = players.find(x => x.id === id);

      if(p) p.group = gruppo;

      render();

    }


    function generaGruppiAuto() {

      const criterio = document.getElementById('split-criterion').value;

      const presenti = players.filter(p => p.status === 'p');


      if (criterio === 'ruolo') {

        const ruoli = ['POR', 'DIF', 'CEN', 'ATT'];

        let toggle = true;

        ruoli.forEach(r => {

          const perRuolo = presenti.filter(p => p.role === r);

          perRuolo.forEach(p => {

            p.group = toggle ? 'rosso' : 'blu';

            toggle = !toggle;

          });

        });

      } else if (criterio === 'voto') {

        const ordinati = [...presenti].sort((a, b) => b.rating - a.rating);

        ordinati.forEach((p, index) => {

          p.group = (index % 2 === 0) ? 'rosso' : 'blu';

        });

      }

      render();

    }


    function resetGruppi() {

      players.forEach(p => p.group = null);

      render();

    }
